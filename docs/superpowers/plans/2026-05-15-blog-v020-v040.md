### Task 1: Compute prev/next in slug page

**Files:**
- Modify: `src/pages/blog/[...slug].astro`

- [ ] **Step 1: Replace `getStaticPaths` with prev/next computation**

Open `src/pages/blog/[...slug].astro`. Replace the entire frontmatter with:

```typescript
---
import { getCollection, type CollectionEntry } from 'astro:content';
import PostLayout from '../../layouts/PostLayout.astro';

export async function getStaticPaths() {
  const posts = (await getCollection('blog', ({ data }) => !data.draft))
    .sort((a, b) => b.data.pubDate.valueOf() - a.data.pubDate.valueOf());

  return posts.map((post, index) => ({
    params: { slug: post.slug },
    props: {
      post,
      prev: posts[index + 1]
        ? { slug: posts[index + 1].slug, title: posts[index + 1].data.title }
        : undefined,
      next: posts[index - 1]
        ? { slug: posts[index - 1].slug, title: posts[index - 1].data.title }
        : undefined,
    },
  }));
}

interface Props {
  post: CollectionEntry<'blog'>;
  prev?: { slug: string; title: string };
  next?: { slug: string; title: string };
}

const { post, prev, next } = Astro.props;
const { Content } = await post.render();
const { title, pubDate, tags } = post.data;
---
```

The template (below the frontmatter) becomes:

```astro

  <Content />
</PostLayout>
```

### Task 4: Create dynamic tag page

**Files:**
- Create: `src/pages/tags/[tag].astro`

- [ ] **Step 1: Create the file**

Create `src/pages/tags/[tag].astro` with:

```astro
---
import { getCollection } from 'astro:content';
import BaseLayout from '../../layouts/BaseLayout.astro';
import PostCard from '../../components/PostCard.astro';

export async function getStaticPaths() {
  const posts = (await getCollection('blog', ({ data }) => !data.draft))
    .sort((a, b) => b.data.pubDate.valueOf() - a.data.pubDate.valueOf());

  const tagMap = new Map<string, typeof posts>();
  for (const post of posts) {
    for (const tag of post.data.tags) {
      if (!tagMap.has(tag)) tagMap.set(tag, []);
      tagMap.get(tag)!.push(post);
    }
  }

  return Array.from(tagMap.entries()).map(([tag, tagPosts]) => ({
    params: { tag },
    props: { tag, posts: tagPosts },
  }));
}

interface Props {
  tag: string;
  posts: Awaited<ReturnType<typeof getCollection<'blog'>>>;
}

const { tag, posts } = Astro.props;
---
<BaseLayout title={`#${tag} — Southern Light`} description={`Posts tagged #${tag}`}>
  <h2 class="section-title">#{tag}</h2>
  <ul class="post-list">
    {posts.map((post) => <PostCard post={post} />)}
  </ul>
</BaseLayout>
```

- [ ] **Step 1: Create the feed file**

Create `src/pages/rss.xml.ts` with:

```typescript
import rss from '@astrojs/rss';
import { getCollection } from 'astro:content';
import type { APIContext } from 'astro';

export async function GET(context: APIContext) {
  const posts = (await getCollection('blog', ({ data }) => !data.draft))
    .sort((a, b) => b.data.pubDate.valueOf() - a.data.pubDate.valueOf());

  return rss({
    title: 'Southern Light',
    description: 'Notes on systems, tools, and side projects.',
    site: context.site!,
    items: posts.map((post) => ({
      title: post.data.title,
      pubDate: post.data.pubDate,
      description: '',
      link: `/blog/${post.slug}/`,
    })),
  });
}
```

```typescript
import { getCollection, type CollectionEntry } from 'astro:content';
import PostLayout from '../../layouts/PostLayout.astro';

function estimateReadingTime(text: string): number {
  const cjk = (text.match(/[　-鿿가-힯豈-﫿]/g) ?? []).length;
  const words = (text.match(/[a-zA-Z]+/g) ?? []).length;
  return Math.max(1, Math.ceil((cjk + words) / 300));
}

export async function getStaticPaths() {
  const posts = (await getCollection('blog', ({ data }) => !data.draft))
    .sort((a, b) => b.data.pubDate.valueOf() - a.data.pubDate.valueOf());

  return posts.map((post, index) => ({
    params: { slug: post.slug },
    props: {
      post,
      prev: posts[index + 1]
        ? { slug: posts[index + 1].slug, title: posts[index + 1].data.title }
        : undefined,
      next: posts[index - 1]
        ? { slug: posts[index - 1].slug, title: posts[index - 1].data.title }
        : undefined,
    },
  }));
}

interface Props {
  post: CollectionEntry<'blog'>;
  prev?: { slug: string; title: string };
  next?: { slug: string; title: string };
}

const { post, prev, next } = Astro.props;
const { Content } = await post.render();
const { title, pubDate, tags } = post.data;
const readingTime = estimateReadingTime(post.body);
---
```
